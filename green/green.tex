  {\Huge\bfseries GREEN'S vs STOKES' THEOREM}\\[10pt]
  {\Large Green’s circulation theorem is Stokes on a plane.}\\[10pt]
  % left
  \begin{minipage}{0.49\textwidth}
  \begin{tcolorbox}[title=GREEN'S THEOREM]
    Planar region $D$ in the $xy$-plane
    [R Green Plot Here]
    \begin{align*}
      \oint_C\left(P\,\mathrm dx+Q\,\mathrm dy\right)=\iint_D\left(\frac{\partial Q}{\partial x}-\frac{\partial P}{\partial y}\right)\,\mathrm dA
    \end{align*}
    Scalar curl integrated over a planar region.
  \end{tcolorbox}
  \end{minipage}
  \hfill
  % right
  \begin{minipage}{0.49\textwidth}
  \begin{tcolorbox}[title=STOKES' THEOREM]
    Surface $S$ bounded by closed curve $C$
    [R Stokes Plot Here]
    \begin{align*}
      \oint_C \boldsymbol{F}\cdot\mathrm d\boldsymbol{r}=\iint_S (\nabla\times\boldsymbol{F})\cdot\mathrm d\boldsymbol{S}
    \end{align*}
    Curl flux over a surface.
  \end{tcolorbox}
  \end{minipage}
  % bottom
  \begin{tcolorbox}[title=Recover Green from Stokes]
  \renewcommand{\tabularxcolumn}[1]{m{#1}}
  \begin{tabularx}{\linewidth}{*{4}{>{\centering\arraybackslash}X|}>{\centering\arraybackslash}X}
  $\boldsymbol F=(P,Q,0)$&$\widehat{\boldsymbol n}=(0,0,1)$&$\boldsymbol F\cdot\mathrm dr=P\mathrm dx+Q\mathrm dy$&$(\nabla\times\boldsymbol F)\cdot\widehat{\boldsymbol n}$&$\mathrm dS=\mathrm dA$
  \end{tabularx}
  On a flat $xy$-surface, Stokes reduces exactly to Green's circulation form.
  \end{tcolorbox}
